cmake_minimum_required(VERSION 3.20)
project(demo C)
set(CMAKE_C_STANDARD 11)
option(FEATURE_GREET "greeting" OFF)
option(FEATURE_MATH  "math"     OFF)
option(FEATURE_STATS "stats"    OFF)
add_executable(app main.c)
if(FEATURE_GREET)
  target_compile_definitions(app PRIVATE FEATURE_GREET)
endif()
if(FEATURE_MATH)
  target_compile_definitions(app PRIVATE FEATURE_MATH)
endif()
if(FEATURE_STATS)
  target_compile_definitions(app PRIVATE FEATURE_STATS)
endif()
